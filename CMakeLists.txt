cmake_minimum_required(VERSION 3.24)
project(RookForge VERSION 1.0.0 LANGUAGES CXX)

option(CHESS_BUILD_GUI "Build the SFML desktop application" OFF)
option(CHESS_BUILD_TESTS "Build tests" ON)
option(CHESS_BUILD_CLI "Build command-line interface" ON)
option(CHESS_FETCH_DEPS "Fetch optional third-party dependencies" OFF)
option(CHESS_SANITIZE "Enable address and undefined-behaviour sanitizers" OFF)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated/rookforge")
configure_file(cmake/rookforge_version.hpp.in generated/rookforge/version.hpp @ONLY)

add_library(rookforge_core src/core/board.cpp src/core/search.cpp src/core/clock.cpp src/core/notation.cpp src/core/game.cpp src/core/session.cpp)
target_include_directories(rookforge_core PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MSVC)
  target_compile_options(rookforge_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(rookforge_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(CHESS_SANITIZE AND NOT MSVC)
  target_compile_options(rookforge_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(rookforge_core PUBLIC -fsanitize=address,undefined)
endif()

if(CHESS_BUILD_CLI)
  add_executable(rookforge_cli src/cli/main.cpp)
  set_target_properties(rookforge_cli PROPERTIES OUTPUT_NAME rookforge_cli)
  target_link_libraries(rookforge_cli PRIVATE rookforge_core)
endif()

if(CHESS_BUILD_TESTS)
  enable_testing()
  if(CHESS_FETCH_DEPS)
    include(FetchContent)
    FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.zip)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
  else()
    find_package(GTest CONFIG QUIET)
  endif()
  if(GTest_FOUND OR CHESS_FETCH_DEPS)
    add_executable(rookforge_tests tests/test_gtest.cpp)
    target_link_libraries(rookforge_tests PRIVATE rookforge_core GTest::gtest_main)
    include(GoogleTest)
    gtest_discover_tests(rookforge_tests)
  else()
    add_executable(rookforge_tests tests/test_engine.cpp)
    target_link_libraries(rookforge_tests PRIVATE rookforge_core)
    add_test(NAME rookforge_tests COMMAND rookforge_tests)
  endif()
endif()

include(GNUInstallDirs)

if(CHESS_BUILD_GUI)
  find_package(SFML 3.0 COMPONENTS Graphics Window System QUIET)
  if(NOT SFML_FOUND)
    message(FATAL_ERROR "CHESS_BUILD_GUI requires SFML 3.0.2. Install SFML or configure with -DCHESS_BUILD_GUI=OFF.")
  endif()
  add_executable(RookForge src/gui/main.cpp)
  set_target_properties(RookForge PROPERTIES OUTPUT_NAME RookForge)
  target_link_libraries(RookForge PRIVATE rookforge_core SFML::Graphics SFML::Window SFML::System)
  target_compile_definitions(RookForge PRIVATE ROOKFORGE_VERSION="${PROJECT_VERSION}")
  install(TARGETS RookForge RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()

if(CHESS_BUILD_CLI)
  install(TARGETS rookforge_cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
include(CPack)
